11. Testing

Shelf has grown past what curl can keep checking. Its tests send requests through the real routes and middleware to a real database, with ktm test, and run in a fraction of a second.

A package for the API

Tests call the code they test, and a test can't call into main.ktm. So the handlers, the middleware and the routes move to a package of the project, api/, and main.ktm keeps what a program does at start: settings, the database, migrations, listening.

shelf/
  ktm.mod
  main.ktm              settings, database, migrations, http.listen
  api/
    api.ktm             module api; handlers, middleware, routes()
    api_test.ktm        module api; the tests
  migrations/

api/api.ktm is the code that leaves main.ktm, unchanged: everything from @json struct Bookmark down to showCollection (the structs, COLUMNS, gNoUser, the token functions and every handler). On top go a module api; line and the imports that code uses; at the bottom, one new function. runMigrate, Settings and main stay in main.ktm. The whole file:

api/api.ktm 
module api;
import "github.com/ozzenerol/ktm-pg";
import crypto, encoding.base64, json, net.http, net.http.middleware, reflect, str, uuid, vec;

/* Shelf's HTTP API: its routes, handlers and middleware. main.ktm reads
 * the settings, opens the database and serves what routes() set up. */

@json struct Bookmark {
    String id;
    String url;
    String title;
    String notes;
    vec.Vec tags;
    String createdAt;
};

/* A page of bookmarks: `next` is the cursor for the page after it */
@json struct Page {
    vec.Vec items;
    @omitEmpty String next;
};

@json struct NewBookmark {
    @required @url @max(2048) u8[] url;
    @max(200) u8[] title;
    @max(2000) u8[] notes;
    vec.Vec tags;
};

/* The columns of a Bookmark, named as its fields */
#define COLUMNS "id, url, title, notes, " \
    "(SELECT coalesce(array_agg(tag ORDER BY tag), '{}') FROM tags WHERE bookmark_id = id) AS tags, to_char(created_at AT TIME ZONE 'UTC', 'YYYY-MM-DD\"T\"HH24:MI:SS\"Z\"') AS \"createdAt\""

/* ---- users and sessions ---- */

@json struct NewUser {
    @required @email @max(254) u8[] email;
    @required @min(10) @max(200) u8[] password;
};

@json struct User {
    String id;
    String email;
};

@json struct Login {
    @required u8[] email;
    @required u8[] password;
};

@json struct Session {
    String token;
    String expiresAt;
};

struct Credentials {
    String id;
    String passwordHash;
};

/* A hash no password matches: a login for an unknown email checks the
 * password against it, so it takes as long as one for a known email */
String gNoUser;

void signup(http.Writer *w, http.Request *r)
{
    pg.Pool *db = r.app;
    NewUser in = http.decodeValid(r);
    String hash = crypto.hashPassword(in.password);     /* Argon2id, on a worker thread */
    defer hash.free();
    String id = uuid.v7().string();
    defer id.free();
    User u;
    try {
        u = pg.queryOne(db, "INSERT INTO users (id, email, password_hash) VALUES ($1, lower($2), $3) RETURNING id, email",
                              id, in.email, hash);
    } catch (Error e) {
        if (e.code != pg.UNIQUE_VIOLATION) { throw(e); }
        w.error(409, "that email has an account");
        return;
    }
    defer u.freeDecoded();
    w.json(201, reflect.of(u));
}

/* The SHA-256 of a token, as the sessions table keeps it */
void tokenHash(u8[] token, u8 *out)
{
    crypto.sha256(token, out[0:32]);
}

void login(http.Writer *w, http.Request *r)
{
    pg.Pool *db = r.app;
    Login in = http.decodeValid(r);
    vec.Vec found = pg.queryAll(db,
        "SELECT id, password_hash AS \"passwordHash\" FROM users WHERE email = lower($1)", in.email);
    defer pg.free(reflect.of(found));
    u8[] hash = found.len ? found.data[0].passwordHash[0:] : gNoUser[0:];
    if (!crypto.verifyPassword(in.password, hash) || !found.len) {
        w.error(401, "wrong email or password");
        return;
    }
    /* 32 random bytes are the token; the database keeps only their hash */
    String raw = crypto.randomBytes(32);
    defer raw.free();
    Session s;
    defer s.freeDecoded();
    s.token = base64.encodeUrl(raw[0:]);
    u8 h[32];
    tokenHash(s.token[0:], h);
    s.expiresAt = pg.queryOne(db,
        "INSERT INTO sessions (token_hash, user_id, expires_at) VALUES ($1, $2, now() + interval '30 days')"
        " RETURNING to_char(expires_at AT TIME ZONE 'UTC', 'YYYY-MM-DD\"T\"HH24:MI:SS\"Z\"')",
        h[0:32], found.data[0].id);
    w.json(201, reflect.of(s));
}

/* Is token a live session? Then the handlers get its user (userOf) */
bool checkToken(u8[] token, http.Request *r)
{
    pg.Pool *db = r.app;
    u8 h[32];
    tokenHash(token, h);
    vec.Vec ids = pg.queryAll(db,
        "SELECT user_id::text FROM sessions WHERE token_hash = $1 AND expires_at > now()", h[0:32]);
    defer pg.free(reflect.of(ids));
    if (!ids.len) { return false; }
    r.setText("userId", ids.data[0][0:]);         /* a copy, kept until the reply */
    return true;
}

/* The signed-in user's id (behind middleware.bearer) */
u8[] userOf(http.Request *r)
{
    return r.getText("userId");
}

void logout(http.Writer *w, http.Request *r)
{
    pg.Pool *db = r.app;
    u8 h[32];
    tokenHash(str.trim(r.header("Authorization")[7:]), h);
    db.exec("DELETE FROM sessions WHERE token_hash = $1", h[0:32]);
    w.status(204);
}

/* ---- collections ---- */

@json struct NewCollection {
    @required @max(100) u8[] name;
    @key("public") bool isPublic;
};

@json struct Collection {
    String id;
    String name;
    @key("public") bool isPublic;
    vec.Vec bookmarks;
};

/* The signed-in user if the request has a good token, else "": for the
 * routes anyone may call */
u8[] maybeUser(http.Request *r)
{
    u8[] auth = str.trim(r.header("Authorization"));
    if (auth.len > 7 && checkToken(str.trim(auth[7:]), r)) { return userOf(r); }
    return "";
}

void newCollection(http.Writer *w, http.Request *r)
{
    pg.Pool *db = r.app;
    NewCollection in = http.decodeValid(r);
    String id = uuid.v7().string();
    defer id.free();
    db.exec("INSERT INTO collections (id, user_id, name, public) VALUES ($1, $2, $3, $4)", id, userOf(r), in.name, in.isPublic);
    Collection c;
    c.id = str.from(id[0:]);
    c.name = str.from(in.name);
    c.isPublic = in.isPublic;
    defer c.freeDecoded();
    w.json(201, reflect.of(c));
}

/* PUT /collections/{id}/bookmarks/{bookmark}: both must be the user's */
void addToCollection(http.Writer *w, http.Request *r)
{
    pg.Pool *db = r.app;
    uuid.UUID a, b;
    if (!uuid.parse(r.param("id"), &a) || !uuid.parse(r.param("bookmark"), &b)) {
        w.error(404, "no such collection or bookmark");
        return;
    }
    i64 mine = pg.queryOne(db, "SELECT count(*) FROM collections c, bookmarks b"
                                    " WHERE c.id = $1 AND b.id = $2 AND c.user_id = $3 AND b.user_id = $3",
                                r.param("id"), r.param("bookmark"), userOf(r));
    if (!mine) {
        w.error(404, "no such collection or bookmark");
        return;
    }
    db.exec("INSERT INTO collection_items (collection_id, bookmark_id) VALUES ($1, $2) ON CONFLICT DO NOTHING",
            r.param("id"), r.param("bookmark"));
    w.status(204);
}

/* GET /collections/{id}: its owner, or anyone when it is public */
void showCollection(http.Writer *w, http.Request *r)
{
    pg.Pool *db = r.app;
    uuid.UUID id;
    if (!uuid.parse(r.param("id"), &id)) {
        w.error(404, "no such collection");
        return;
    }
    /* someone else's private collection is a 404, not a 403: it doesn't
     * say that it exists */
    vec.Vec found = pg.queryAll(db,
        "SELECT id, name, public FROM collections WHERE id = $1 AND (public OR user_id::text = $2)",
        r.param("id"), maybeUser(r));
    defer pg.free(reflect.of(found));
    if (!found.len) {
        w.error(404, "no such collection");
        return;
    }
    Collection *c = &found.data[0];
    c.bookmarks = pg.queryAll(db,
        "SELECT " COLUMNS " FROM bookmarks WHERE id IN"
        " (SELECT bookmark_id FROM collection_items WHERE collection_id = $1) ORDER BY id DESC", r.param("id"));
    w.json(200, reflect.of(*c));
}

/* GET /bookmarks?tag=go&q=fast+servers&limit=20&after=: newest first */
void list(http.Writer *w, http.Request *r)
{
    pg.Pool *db = r.app;
    i64 limit = 20;
    if (r.hasQuery("limit") && (!str.toInt(r.query("limit"), &limit) || limit < 1 || limit > 100)) {
        w.error(400, "limit is 1 to 100");
        return;
    }
    uuid.UUID after;
    u8[] cursor = r.query("after");
    if (cursor.len && !uuid.parse(cursor, &after)) {
        w.error(400, "after isn't a cursor from this API");
        return;
    }
    Page page;
    defer page.freeDecoded();
    /* one more than asked for: if it comes back, there is a next page */
    page.items = pg.queryAll(db,
        "SELECT " COLUMNS " FROM bookmarks"
        " WHERE user_id = $1"
        "   AND ($2 = '' OR id IN (SELECT bookmark_id FROM tags WHERE tag = $2))"
        "   AND ($3 = '' OR search @@ websearch_to_tsquery('simple', $3))"
        "   AND ($4 = '' OR id < $4::uuid)"
        " ORDER BY id DESC LIMIT $5",
        userOf(r), r.query("tag"), r.query("q"), cursor, limit + 1);
    if (page.items.len > limit) {
        Bookmark extra = page.items.removeAt(limit);
        pg.free(reflect.of(extra));
        page.next = str.from(page.items.data[limit - 1].id[0:]);
    }
    w.json(200, reflect.of(page));
}

/* A tag is 1 to 30 lowercase letters, digits and dashes */
bool goodTag(u8[] t)
{
    if (t.len < 1 || t.len > 30) { return false; }
    for (auto c : t) {
        if (!((c >= 'a' && c <= 'z') || (c >= '0' && c <= '9') || c == '-')) { return false; }
    }
    return true;
}

/* The id in the path, or false (and a 404) when it isn't a UUID */
bool pathId(http.Writer *w, http.Request *r, uuid.UUID *id)
{
    if (uuid.parse(r.param("id"), id)) { return true; }
    w.error(404, "no such bookmark");
    return false;
}

void get(http.Writer *w, http.Request *r)
{
    pg.Pool *db = r.app;
    uuid.UUID id;
    if (!pathId(w, r, &id)) { return; }
    Bookmark b;
    try {
        b = pg.queryOne(db, "SELECT " COLUMNS " FROM bookmarks WHERE id = $1 AND user_id = $2",
                                  r.param("id"), userOf(r));
    } catch (Error e) {
        if (e.code != pg.ERR_NO_ROWS) { throw(e); }
        w.error(404, "no such bookmark");
        return;
    }
    defer b.freeDecoded();
    w.json(200, reflect.of(b));
}

void create(http.Writer *w, http.Request *r)
{
    pg.Pool *db = r.app;
    NewBookmark in = http.decodeValid(r);
    if (in.tags.len > 10) {
        w.error(422, "at most 10 tags");
        return;
    }
    for (auto t : in.tags) {
        if (!goodTag(t[0:])) {
            w.error(422, "a tag is 1 to 30 lowercase letters, digits and dashes");
            return;
        }
    }
    String id = uuid.v7().string();
    defer id.free();
    /* the bookmark and its tags together, or neither */
    pg.Tx tx = db.begin();
    defer tx.rollback();
    try {
        tx.exec("INSERT INTO bookmarks (id, user_id, url, title, notes) VALUES ($1, $2, $3, $4, $5)",
                id, userOf(r), in.url, in.title.len ? in.title : in.url, in.notes);
    } catch (Error e) {
        if (e.code != pg.UNIQUE_VIOLATION) { throw(e); }
        w.error(409, "that link is already on the shelf");
        return;
    }
    tx.exec("INSERT INTO tags (bookmark_id, tag) SELECT $1, unnest($2::text[]) ON CONFLICT DO NOTHING", id, in.tags);
    Bookmark b = pg.queryOne(&tx, "SELECT " COLUMNS " FROM bookmarks WHERE id = $1", id);
    defer b.freeDecoded();
    tx.commit();
    w.json(201, reflect.of(b));
}

void remove(http.Writer *w, http.Request *r)
{
    pg.Pool *db = r.app;
    uuid.UUID id;
    if (!pathId(w, r, &id)) { return; }
    if (db.exec("DELETE FROM bookmarks WHERE id = $1 AND user_id = $2", r.param("id"), userOf(r)) == 0) {
        w.error(404, "no such bookmark");
        return;
    }
    w.status(204);
}

/* Shelf's routes and middleware on mux. webOrigin: the web app that may
 * call it from a browser ("": none); trustedProxies: the load balancer's
 * addresses ("": none). */
pub void routes(http.Mux *mux, u8[] webOrigin, u8[] trustedProxies)
{
    if (!gNoUser.len) {
        String noise = uuid.v4().string();
        gNoUser = crypto.hashPassword(noise[0:]);
        noise.free();
    }

    if (trustedProxies.len) { middleware.realIp(mux, trustedProxies); }
    middleware.requestId(mux);
    if (webOrigin.len) {
        middleware.CorsOptions cors;
        cors.origins = webOrigin;
        cors.expose = "X-Request-Id";
        middleware.cors(mux, &cors);
    }
    middleware.rateLimit(mux, 300, 60);            /* any client: 300 requests a minute */

    mux.handle("POST /users", &signup);
    http.Mux *logins = mux.group("/sessions");
    middleware.rateLimit(logins, 5, 60);            /* guessing passwords: 5 tries a minute */
    logins.handle("POST", &login);
    http.Mux *session = mux.group("/sessions");
    middleware.bearer(session, &checkToken);
    session.handle("DELETE", &logout);

    /* every bookmark route needs a signed-in user */
    http.Mux *bookmarks = mux.group("/bookmarks");
    middleware.bearer(bookmarks, &checkToken);
    bookmarks.handle("GET", &list);
    bookmarks.handle("POST", &create);
    bookmarks.handle("GET /{id}", &get);
    bookmarks.handle("DELETE /{id}", &remove);

    http.Mux *collections = mux.group("/collections");
    middleware.bearer(collections, &checkToken);
    collections.handle("POST", &newCollection);
    collections.handle("PUT /{id}/bookmarks/{bookmark}", &addToCollection);
    mux.handle("GET /collections/{id}", &showCollection);      /* outside the group: anyone may ask */
}

The package is imported by the project's path from ktm.mod (module example.com/shelf) and its directory. main.ktm is now this, the whole file:

import "github.com/ozzenerol/ktm-pg";
import "github.com/ozzenerol/ktm-migrate";
import "github.com/ozzenerol/ktm-migrate/postgres";
import "example.com/shelf/api";
import config, flag, log, net.http, os, reflect, str, vec;

/* shelf migrate up | down | status */
i32 runMigrate(migrate.Migrations *ms, migrate.Driver *d, u8[] what)
{
    if (str.equals(what, "up")) {
        printf("%d migrations ran\n", ms.up(d));
    } else if (str.equals(what, "down")) {
        printf("%d migration reverted\n", ms.down(d, 1));
    } else if (str.equals(what, "status")) {
        vec.Vec all = ms.status(d);
        for (auto s : all) {
            printf("%4d  %-22s %s\n", s.version, s.name, s.applied ? "applied" : "pending");
        }
        all.free();
    } else {
        log.error("shelf migrate: up, down or status, not `%s`", what);
        return 2;
    }
    return 0;
}

struct Settings {
    i64 port;
    bool accessLog;
    @required u8[] databaseUrl;
    u8[] webOrigin;             /* the web app allowed to call the API from a browser */
    u8[] trustedProxies;        /* the load balancer's addresses, when there is one */
};

Settings gSettings;

i32 main(i32 argc, u8 **argv)
{
    gSettings.port = 8080;
    config.loadDotenv(".env");
    try {
        config.fromEnv(reflect.of(gSettings), "SHELF_");
    } catch (Error e) {
        log.error("%s", e.message);     /* every bad or missing variable, by name */
        os.exit(2);
    }

    flag.Set *flags = flag.new("shelf", "a bookmarks API");
    defer flags.free();
    i64 *port = flags.integer("port", gSettings.port, "the port to listen on", 'p');
    bool *accessLog = flags.boolean("access-log", gSettings.accessLog, "log every request");
    flag.Set *migrateCmd = flags.command("migrate", "up, down or status, then exit");
    flags.parseOrExit(argc, argv);

    /* the database: a pool of connections shared by every thread */
    pg.Pool *db = pg.open(gSettings.databaseUrl);
    defer db.close();
    migrate.Migrations ms = migrate.load("migrations");
    defer ms.free();
    migrate.Driver d = postgres.driverPool(db);
    if (flags.chosen() == migrateCmd) {
        return runMigrate(&ms, &d, migrateCmd.arg(0));
    }
    i64 ran = ms.up(&d);
    log.info("database at version %d (%d migrations ran)", migrate.version(&d), ran);

    http.Mux mux;
    api.routes(&mux, gSettings.webOrigin, gSettings.trustedProxies);

    http.Options o;
    o.accessLog = *accessLog;
    log.info("Shelf listening on http://localhost:%d", *port);
    return http.listen((i32)*port, &mux, db, &o);
}

crypto, encoding.base64, json, uuid and net.http.middleware leave its imports with the code that used them.

The tests

A test file is named *_test.ktm and starts with the same module api; line, so its tests see every function of the package, private ones too. Each void testXxx(testing.T *t) is a test.

module api;
import "github.com/ozzenerol/ktm-pg";
import "github.com/ozzenerol/ktm-migrate";
import "github.com/ozzenerol/ktm-migrate/postgres";
import net.http, proc, str, testing;

/* The API's tests, through the real routes and middleware, against a
 * database of their own: SHELF_TEST_DATABASE_URL (they are skipped
 * without it). Every test starts from empty tables. */

http.Mux gMux;
pg.Pool *gDb;
i64 gClient = 0;        /* each test is another client, with its own rate limits */

/* The mux and the database, once; empty tables for every test */
bool setup(testing.T *t)
{
    if (!gDb) {
        String url = proc.getenv("SHELF_TEST_DATABASE_URL");
        defer url.free();
        if (!url.len) { t.skip("SHELF_TEST_DATABASE_URL isn't set"); }
        gDb = pg.open(url[0:]);
        migrate.Migrations ms = migrate.load("../migrations");
        migrate.Driver d = postgres.driverPool(gDb);
        ms.up(&d);
        ms.free();
        routes(&gMux, "", "");
    }
    gDb.exec("TRUNCATE users, bookmarks CASCADE");
    gClient++;
    return true;
}

/* A request through the routes: the reply, yours to free */
http.Writer send(u8[] method, u8[] target, u8[] body = "", u8[] token = "")
{
    http.Request r = http.testRequest(method, target, body);
    r.app = gDb;
    String addr = str.format("192.0.2.%d", gClient);
    r.setRemoteAddr(addr[0:]);
    addr.free();
    String auth = str.format("Bearer %s", token);
    if (token.len) { r.addHeader("Authorization", auth[0:]); }
    http.Writer w = http.recorder();
    gMux.serve(&w, &r);
    r.free();
    auth.free();
    return w;
}

/* A new user's session token (yours to free) */
String signIn(testing.T *t, u8[] email)
{
    String body = str.format("{\"email\": \"%s\", \"password\": \"a long password\"}", email);
    defer body.free();
    http.Writer w = send("POST", "/users", body[0:]);
    t.equal(w.code, 201);
    w.free();
    w = send("POST", "/sessions", body[0:]);
    t.equal(w.code, 201);
    i64 at = str.find(w.body[0:], "\"token\":\"") + 9;
    String token = str.from(w.body[at:at + 43]);
    w.free();
    return token;
}

void testSignupRules(testing.T *t)
{
    setup(t);
    http.Writer w = send("POST", "/users", "{\"email\": \"not an email\", \"password\": \"short\"}");
    t.equal(w.code, 422);
    t.check(str.find(w.body[0:], "\"field\":\"email\"") > 0 && str.find(w.body[0:], "\"field\":\"password\"") > 0,
            "both fields named: %s", w.body);
    w.free();
}

void testLogin(testing.T *t)

void testUsersSeeOnlyTheirOwn(testing.T *t)
{
    setup(t);
    String ada = signIn(t, "ada@example.com");
    String bob = signIn(t, "bob@example.com");
    defer ada.free();
    defer bob.free();
    http.Writer w = send("POST", "/bookmarks", "{\"url\": \"https://go.dev\", \"tags\": [\"go\"]}", ada[0:]);
    t.equal(w.code, 201);
    w.free();
    w = send("GET", "/bookmarks", "", bob[0:]);
    t.equal(w.body, "{\"items\":[]}");
    w.free();
    w = send("GET", "/bookmarks?tag=go", "", ada[0:]);
    t.check(str.find(w.body[0:], "https://go.dev") > 0, "ada's list: %s", w.body);
    w.free();
}

void testDuplicateLink(testing.T *t)
{
    setup(t);
    String ada = signIn(t, "ada@example.com");
    defer ada.free();
    http.Writer w = send("POST", "/bookmarks", "{\"url\": \"https://go.dev\"}", ada[0:]);
    w.free();
    w = send("POST", "/bookmarks", "{\"url\": \"https://go.dev\"}", ada[0:]);
    t.equal(w.code, 409);
    w.free();
}
  • http.testRequest and http.recorder() make a request and catch the reply without a socket; gMux.serve runs them through the same routes and middleware as the server. r.app is what http.listen would have set.
  • The tests use a database of their own, named by SHELF_TEST_DATABASE_URL, migrated once and emptied before every test. Without it they are skipped, not failed, so ktm test works on a machine with no database.
  • Each test is another client address (setRemoteAddr), so the login rate limit of chapter 10 doesn't carry over from one test to the next.
  • t.equal(got, want) compares numbers and texts and says both when they differ; t.check(ok, "...") takes any condition and a message. A failed check doesn't stop the test: every failure is listed.
