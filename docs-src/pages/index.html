Install (Linux x86_64; arm64 and macOS zips are on the same page) 
curl -LO https://github.com/ozzenerol/ktm-lang/releases/download/{VERSION}/ktm-{VERSION}-linux-x86_64.zip
unzip ktm-{VERSION}-linux-x86_64.zip && cd ktm-{VERSION}-linux-x86_64 && sudo ./install.sh
hello.ktm 
import net.http;

void hello(http.Writer *w, http.Request *r) {
    u8[] name = r.query("name");
    if !name.len { name = "world"; }
    w.text(200, "hello, %s\n", name);
}

i32 main() {
    http.Mux mux;
    mux.handle("GET /", &hello);
    return http.listen(8080, &mux);
}
$ ktm hello.ktm -o hello && ./hello &
$ curl 'localhost:8080/?name=ada'
hello, ada
