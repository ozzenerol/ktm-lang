    LibraryReference: csv

    CSV

    csv.next walks CSV text without copying it: each field, then the whole row, then csv.EOF.

    Walking a text#

    import csv;

void main()
{
  u8[] text = "name,age\nann,41\nbob,29\n";
  csv.Parser p;
  csv.Item it;
  i64 kind;
  csv.init(&p, text);
  while ((kind = csv.next(&p, &it)) != csv.EOF) {
    if (kind == csv.FIELD) { printf("[%s]", it.text); }
    else                   { printf(" <- row %d\n", it.idx); }
  }
}

    [name][age] <- row 0
[ann][41] <- row 1
[bob][29] <- row 2
    • A csv.Item has idx (the field's number in its row, or the row's number) and text, a u8[] view of the input.
    • csv.init takes any u8[]: text[0:n] parses part of a buffer. The input must outlive the parser.

    Quotes are kept#

    The views are the raw text: a quoted field keeps its quotes and doubled "". Commas and newlines inside quotes don't split. A row's text includes its line ending.

    import csv;

void main()
{
  u8[] text = "name,quote\r\nann,\"say \"\"hi\"\", ok\"\nbob,\n";
  csv.Parser p;
  csv.Item it;
  i64 kind;
  csv.init(&p, text);
  while ((kind = csv.next(&p, &it)) != csv.EOF) {
    if (kind == csv.FIELD) { printf("field %d: [%s]\n", it.idx, it.text); }
    else                   { printf("row %d: %d bytes\n", it.idx, it.text.len); }
  }
}

    field 0: [name]
field 1: [quote]
row 0: 12 bytes
field 0: [ann]
field 1: ["say ""hi"", ok"]
row 1: 21 bytes
field 0: [bob]
field 1: []
row 2: 5 bytes

    Unquoting a field#

    Unquoting is up to you. A small helper:

    import csv, str;

String unquote(u8[] f)
{
  if (f.len < 2 || f[0] != '"') { return str.from(f); }
  String out = str.withCap(f.len);
  for (i64 i = 1; i < f.len - 1; i++) {
    out.appendByte(f[i]);
    if (f[i] == '"') { i++; }        // "" is one quote
  }
  return out;
}

void main()
{
  u8[] text = "id,comment\n7,\"fine, \"\"really\"\"\"\n";
  csv.Parser p;
  csv.Item it;
  csv.init(&p, text);
  while (csv.next(&p, &it) != csv.EOF) {
    if (it.idx == 1 && it.text.len && it.text[0] == '"') {
      ad String v = unquote(it.text);
      printf("%s\n", v);
    }
  }
}

    fine, "really"

    Summing a column#

    import cstr, csv, str;

void main()
{
  u8[] text = "item,price\npen,3\nbook,12\nbag,25\n";
  csv.Parser p;
  csv.Item it;
  i64 kind, total = 0;
  bool header = true;
  csv.init(&p, text);
  while ((kind = csv.next(&p, &it)) != csv.EOF) {
    if (kind == csv.ROW) { header = false; continue; }
    if (!header && it.idx == 1) {
      ad String n = str.from(it.text);
      total += cstr.toInt(n.cstr());
    }
  }
  printf("total %d\n", total);
}

    total 40
