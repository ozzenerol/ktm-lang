    A quick tour

    The language in ten short programs. Each one links to the page that explains it.

    1. Hello#

    void main()
{
  printf("Hello, %s! %d + %d = %d\n", "ktm", 2, 3, 2 + 3);
}

    Hello, ktm! 2 + 3 = 5

    Program basics

    2. Types and variables#

    const i64 LIMIT = 3;

void main()
{
  i64 count = 0;
  f64 ratio = 0.5;
  u8 letter = 'k';
  bool ready = true;
  auto name = "ktm";            // u8[4]
  for (i64 i = 0; i < LIMIT; i++) { count += i; }
  printf("%d %.1f %c %d %s\n", count, ratio, letter, ready, name);
}

    3 0.5 k 1 ktm

    Types and literals, Variables

    3. Functions with defaults#

    u8 *greet(u8 *who = "world", bool loud = false)
{
  if (loud) { printf("HELLO, %s!\n", who); } else { printf("hello, %s\n", who); }
  return who;
}

void main()
{
  greet();
  greet("ada");
  greet(, true);
}

    hello, world
hello, ada
HELLO, world!

    Functions

    4. Slices#

    f64 mean(f64[] xs)
{
  f64 total = 0;
  for (auto x : xs) { total += x; }
  return total / xs.len;
}

void main()
{
  f64 temps[] = {18.5, 21.0, 16.5, 20.0};
  printf("%.2f %.2f\n", mean(temps), mean(temps[1:3]));
}

    19.00 18.75

    Slices

    5. Strings#

    import str;

void main()
{
  ad String s = str.format("%s-%d", "build", 42);
  s.append(".log");
  ad u8[][] parts = str.split(s, "-");
  printf("%s has %d bytes, %d parts, ends with .log: %d\n",
         s, s.len, parts.len, s.endsWith(".log"));
}

    build-42.log has 12 bytes, 2 parts, ends with .log: 1

    Strings, the str module

    6. Structs and methods#

    struct Rect { f64 w, h; };

f64 Rect.area() { return self.w * self.h; }
void Rect.scale(f64 k) { self.w *= k; self.h *= k; }

void main()
{
  Rect r = {2, 3};
  r.scale(2);
  printf("%.0f x %.0f = %.0f\n", r.w, r.h, r.area());
}

    4 x 6 = 24

    Structs, Methods

    7. Errors and clean-up#

    enum Err { tooBig };

i64 checked(i64 n)
{
  defer printf("checked %d\n", n);
  if (n > 100) { throw(Err.tooBig, "%d is over 100", n); }
  return n;
}

void main()
{
  try {
    checked(5);
    checked(500);
  } catch (Error e) {
    printf("error: %s\n", e.message);
  }
}

    checked 5
checked 500
error: 500 is over 100

    Exceptions, defer

    8. Containers#

    import map, vec;

void main()
{
  ad map.Map<u8[], i64> counts = map.new<u8[], i64>();
  u8[] words[] = {"a", "b", "a", "c", "a"};
  for (auto w : words) { counts.set(w, counts.getOr(w, 0) + 1); }

  ad vec.Vec<i64> sizes = counts.values();
  sizes.sortAsc();
  printf("a=%d, %d distinct, smallest count %d\n", counts.getOr("a", 0), counts.len, sizes.get(0));
}

    a=3, 3 distinct, smallest count 1

    Hash maps, Vectors

    9. Generics#

    struct Pair<A, B> { A first; B second; };

T clamp<T>(T v, T lo, T hi) { return v < lo ? lo : v > hi ? hi : v; }

void main()
{
  Pair<u8 *, i64> p = {"port", clamp(70000, 1, 65535)};
  printf("%s %d %.1f\n", p.first, p.second, clamp(1.5, 0.0, 1.0));
}

    port 65535 1.0

    Generics

    10. JSON in and out#

    import json, reflect;

@json
struct Server { String host; i64 port; bool tls; };

void main()
{
  Server s = json.decode<Server>("{\"host\": \"example.com\", \"port\": 443, \"tls\": true}");
  defer s.freeDecoded();
  s.port = 8443;
  ad String out = json.encode(reflect.of(s));
  printf("%s\n", out);
}

    {"host":"example.com","port":8443,"tls":true}
