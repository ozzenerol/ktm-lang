    LibraryReference: thread atomic

    Threads and atomics

    POSIX threads straight from libc, a thread pool in thread, and lock-free integers in atomic.

    Starting threads#

    libc.pthread_create takes a void *f(void *arg). Pass &th to create and th to join.

    import libc;

i64 sums[4];

void *sum(void *arg)
{
  i64 part = (i64)arg, total = 0;
  for (i64 i = part * 250 + 1; i <= (part + 1) * 250; i++) { total += i; }
  sums[part] = total;          // each thread writes its own slot
  return null;
}

void main()
{
  libc.pthread_t *th[4];
  for (i64 i = 0; i < 4; i++) { libc.pthread_create(&th[i], null, &sum, (void *)i); }
  i64 total = 0;
  for (i64 i = 0; i < 4; i++) {
    libc.pthread_join(th[i]);
    total += sums[i];
  }
  printf("1 + 2 + ... + 1000 = %d\n", total);
}

    1 + 2 + ... + 1000 = 500500

    A mutex#

    import libc, map;

libc.pthread_mutex_t gLock;
map.Map<i64, i64> gCounts;

void *work(void *arg)
{
  for (i64 i = 0; i < 1000; i++) {
    libc.pthread_mutex_lock(&gLock);
    i64 *c = gCounts.ref(i % 10);
    if (c) { *c += 1; } else { gCounts.set(i % 10, 1); }
    libc.pthread_mutex_unlock(&gLock);
  }
  return null;
}

void main()
{
  libc.pthread_mutex_init(&gLock, null);
  gCounts = map.new<i64, i64>();
  libc.pthread_t *th[4];
  for (i64 i = 0; i < 4; i++) { libc.pthread_create(&th[i], null, &work, null); }
  for (i64 i = 0; i < 4; i++) { libc.pthread_join(th[i]); }
  printf("%d keys, key 3 counted %d times\n", gCounts.len, gCounts.getOr(3, 0));
  gCounts.free();
}

    10 keys, key 3 counted 400 times

    Condition variables (pthread_cond_*) are declared too.

    A thread pool#

    CallDoes
    thread.poolNew(n)Starts n workers (null if n <= 0 or a thread can't start)
    thread.poolEnqueue(pool, argv, &fn)Runs fn(pool.privData, argv) on a worker
    thread.poolWait(pool)Waits until the queue is empty (never from inside a job)
    thread.poolRelease(pool)Runs what is left and frees the pool
     
    import atomic, thread;

i64 gDone = 0;

void square(void *priv, i64 *n)
{
  *n = *n * *n;
  atomic.add(&gDone, 1);
}

void main()
{
  thread.Pool *pool = thread.poolNew(3);
  i64 nums[8] = {1, 2, 3, 4, 5, 6, 7, 8};
  for (i64 i = 0; i < 8; i++) { thread.poolEnqueue(pool, &nums[i], &square); }
  thread.poolWait(pool);
  thread.poolRelease(pool);
  printf("%d jobs: %d %d %d\n", atomic.load(&gDone), nums[0], nums[3], nums[7]);
}

    8 jobs: 1 16 64

    Atomics#

    Change an integer several threads share, without a mutex: a counter, a flag, a lock-free swap.

    import atomic, libc;

i64 gHits = 0;

void *worker(void *unused)
{
  for (i64 i = 0; i < 100000; i++) { atomic.add(&gHits, 1); }
  return null;
}

void main()
{
  libc.pthread_t *th[4];
  for (i64 i = 0; i < 4; i++) { libc.pthread_create(&th[i], null, &worker, null); }
  for (i64 i = 0; i < 4; i++) { libc.pthread_join(th[i]); }
  printf("%d hits\n", atomic.load(&gHits));
}

    400000 hits
    CallDoes
    atomic.add(&n, v), subAdds or subtracts; returns the value before
    atomic.and, or, xorA bit operation; returns the value before
    atomic.swap(&n, v)Stores; returns the old value
    atomic.load(&n), store(&n, v)Reads, writes
    atomic.cas(&n, expected, desired)Stores desired only if n is expected; says whether it did
    loadPtr, storePtr, swapPtr, casPtrThe same for a void *
    atomic.fence()A full memory barrier
     
    import atomic;

void main()
{
  i64 flags = 0b0101;
  i64 before = atomic.or(&flags, 0b0010);
  printf("or: %d -> %d\n", before, flags);
  bool swapped = atomic.cas(&flags, 7, 0);    // flags is 7: now 0
  bool again = atomic.cas(&flags, 7, 1);      // flags isn't 7 any more
  printf("cas: %d %d, flags %d\n", swapped, again, flags);

  u32 small = 10;
  atomic.sub(&small, 3);                      // the pointer picks the width
  printf("%d\n", small);
}

    or: 5 -> 7
cas: 1 0, flags 0
7
    • Each takes an i64 *, i32 *, u64 * or u32 *; values are passed as i64.
    • Every operation is sequentially consistent: all threads see them in one order.
    • The integer must be aligned to its size, as variables, fields and mem.alloc'd memory are.
    • For anything bigger than one integer, a mutex is simpler.

    Threads and the rest of ktm#

    • Each thread has its own try/catch chain and its own coroutines.
    • An uncaught error in any thread ends the program with exit code 1.
    • For a server, coro.serve runs a coroutine loop per core and is usually simpler than threads by hand.
