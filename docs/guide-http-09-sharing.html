    9. Sharing

    Users group bookmarks into collections, and a public collection can be read by anyone with its link, signed in or not. That needs a route with two audiences, and permission checks that don't leak what exists.

    The migration#

    migrations/005_collections.up.sql:

    CREATE TABLE collections (
    id         uuid PRIMARY KEY,
    user_id    uuid NOT NULL REFERENCES users (id) ON DELETE CASCADE,
    name       text NOT NULL,
    public     boolean NOT NULL DEFAULT false,
    created_at timestamptz NOT NULL DEFAULT now()
);

CREATE TABLE collection_items (
    collection_id uuid NOT NULL REFERENCES collections (id) ON DELETE CASCADE,
    bookmark_id   uuid NOT NULL REFERENCES bookmarks (id) ON DELETE CASCADE,
    PRIMARY KEY (collection_id, bookmark_id)
);

    migrations/005_collections.down.sql undoes it, newest first:

    DROP TABLE collection_items;
DROP TABLE collections;

    The types#

    Under the users and sessions code:

    @json struct NewCollection {
    @required @max(100) u8[] name;
    @key("public") bool isPublic;
};

@json struct Collection {
    String id;
    String name;
    @key("public") bool isPublic;
    vec.Vec<Bookmark> bookmarks;
};
    • public is the column and the JSON key, but a reserved word in ktm, so the field is isPublic with @key("public"): JSON and ktm-pg both read it.

    Routes for owners, a route for everyone#

    In main, after the bookmark routes:

        http.Mux *collections = mux.group("/collections");
    middleware.bearer(collections, &checkToken);
    collections.handle("POST", &newCollection);
    collections.handle("PUT /{id}/bookmarks/{bookmark}", &addToCollection);
    mux.handle("GET /collections/{id}", &showCollection);      /* outside the group: anyone may ask */

    Creating and filling collections is in the group, behind the token check. Reading one is outside it: the handler looks at the token itself, when there is one.

    /* The signed-in user if the request has a good token, else "": for the
 * routes anyone may call */
u8[] maybeUser(http.Request *r)
{
    u8[] auth = str.trim(r.header("Authorization"));
    if (auth.len > 7 && checkToken(str.trim(auth[7:]), r)) { return userOf(r); }
    return "";
}

    Creating one#

    void newCollection(http.Writer *w, http.Request *r)
{
    pg.Pool *db = r.app;
    NewCollection in = http.decodeValid<NewCollection>(r);
    String id = uuid.v7().string();
    defer id.free();
    db.exec("INSERT INTO collections (id, user_id, name, public) VALUES ($1, $2, $3, $4)", id, userOf(r), in.name, in.isPublic);
    Collection c;
    c.id = str.from(id[0:]);
    c.name = str.from(in.name);
    c.isPublic = in.isPublic;
    defer c.freeDecoded();
    w.json(201, reflect.of(c));
}

    The reply is built from what was inserted, so it needs no second query: a new collection has no bookmarks yet.

    Permissions in the query#

    /* GET /collections/{id}: its owner, or anyone when it is public */
void showCollection(http.Writer *w, http.Request *r)
{
    pg.Pool *db = r.app;
    uuid.UUID id;
    if (!uuid.parse(r.param("id"), &id)) {
        w.error(404, "no such collection");
        return;
    }
    /* someone else's private collection is a 404, not a 403: it doesn't
     * say that it exists */
    vec.Vec<Collection> found = pg.queryAll<Collection>(db,
        "SELECT id, name, public FROM collections WHERE id = $1 AND (public OR user_id::text = $2)",
        r.param("id"), maybeUser(r));
    defer pg.free(reflect.of(found));
    if (!found.len) {
        w.error(404, "no such collection");
        return;
    }
    Collection *c = &found.data[0];
    c.bookmarks = pg.queryAll<Bookmark>(db,
        "SELECT " COLUMNS " FROM bookmarks WHERE id IN"
        " (SELECT bookmark_id FROM collection_items WHERE collection_id = $1) ORDER BY id DESC", r.param("id"));
    w.json(200, reflect.of(*c));
}
    • The permission rule is part of the WHERE: public OR user_id = me. There is no "load it, then check" step to forget.
    • Someone else's private collection gets the same 404 as one that doesn't exist. A 403 would confirm it exists.
    /* PUT /collections/{id}/bookmarks/{bookmark}: both must be the user's */
void addToCollection(http.Writer *w, http.Request *r)
{
    pg.Pool *db = r.app;
    uuid.UUID a, b;
    if (!uuid.parse(r.param("id"), &a) || !uuid.parse(r.param("bookmark"), &b)) {
        w.error(404, "no such collection or bookmark");
        return;
    }
    i64 mine = pg.queryOne<i64>(db, "SELECT count(*) FROM collections c, bookmarks b"
                                    " WHERE c.id = $1 AND b.id = $2 AND c.user_id = $3 AND b.user_id = $3",
                                r.param("id"), r.param("bookmark"), userOf(r));
    if (!mine) {
        w.error(404, "no such collection or bookmark");
        return;
    }
    db.exec("INSERT INTO collection_items (collection_id, bookmark_id) VALUES ($1, $2) ON CONFLICT DO NOTHING",
            r.param("id"), r.param("bookmark"));
    w.status(204);
}

    Adding checks that the collection and the bookmark are the user's in one query: otherwise Bob could put Ada's private bookmark in his public collection and read it there.

    The whole program#

    The program at the end of this chapter, to check yours against:

    main.ktm 
    import "github.com/ozzenerol/ktm-pg";
import "github.com/ozzenerol/ktm-migrate";
import "github.com/ozzenerol/ktm-migrate/postgres";
import config, crypto, encoding.base64, flag, json, log, net.http, net.http.middleware, os, reflect, str, uuid, vec;

@json struct Bookmark {
    String id;
    String url;
    String title;
    String notes;
    vec.Vec<String> tags;
    String createdAt;
};

/* A page of bookmarks: `next` is the cursor for the page after it */
@json struct Page {
    vec.Vec<Bookmark> items;
    @omitEmpty String next;
};

@json struct NewBookmark {
    @required @url @max(2048) u8[] url;
    @max(200) u8[] title;
    @max(2000) u8[] notes;
    vec.Vec<String> tags;
};

/* The columns of a Bookmark, named as its fields */
#define COLUMNS "id, url, title, notes, " \
    "(SELECT coalesce(array_agg(tag ORDER BY tag), '{}') FROM tags WHERE bookmark_id = id) AS tags, to_char(created_at AT TIME ZONE 'UTC', 'YYYY-MM-DD\"T\"HH24:MI:SS\"Z\"') AS \"createdAt\""

/* ---- users and sessions ---- */

@json struct NewUser {
    @required @email @max(254) u8[] email;
    @required @min(10) @max(200) u8[] password;
};

@json struct User {
    String id;
    String email;
};

@json struct Login {
    @required u8[] email;
    @required u8[] password;
};

@json struct Session {
    String token;
    String expiresAt;
};

struct Credentials {
    String id;
    String passwordHash;
};

/* A hash no password matches: a login for an unknown email checks the
 * password against it, so it takes as long as one for a known email */
String gNoUser;

void signup(http.Writer *w, http.Request *r)
{
    pg.Pool *db = r.app;
    NewUser in = http.decodeValid<NewUser>(r);
    String hash = crypto.hashPassword(in.password);     /* Argon2id, on a worker thread */
    defer hash.free();
    String id = uuid.v7().string();
    defer id.free();
    User u;
    try {
        u = pg.queryOne<User>(db, "INSERT INTO users (id, email, password_hash) VALUES ($1, lower($2), $3) RETURNING id, email",
                              id, in.email, hash);
    } catch (Error e) {
        if (e.code != pg.UNIQUE_VIOLATION) { throw(e); }
        w.error(409, "that email has an account");
        return;
    }
    defer u.freeDecoded();
    w.json(201, reflect.of(u));
}

/* The SHA-256 of a token, as the sessions table keeps it */
void tokenHash(u8[] token, u8 *out)
{
    crypto.sha256(token, out[0:32]);
}

void login(http.Writer *w, http.Request *r)
{
    pg.Pool *db = r.app;
    Login in = http.decodeValid<Login>(r);
    vec.Vec<Credentials> found = pg.queryAll<Credentials>(db,
        "SELECT id, password_hash AS \"passwordHash\" FROM users WHERE email = lower($1)", in.email);
    defer pg.free(reflect.of(found));
    u8[] hash = found.len ? found.data[0].passwordHash[0:] : gNoUser[0:];
    if (!crypto.verifyPassword(in.password, hash) || !found.len) {
        w.error(401, "wrong email or password");
        return;
    }
    /* 32 random bytes are the token; the database keeps only their hash */
    String raw = crypto.randomBytes(32);
    defer raw.free();
    Session s;
    defer s.freeDecoded();
    s.token = base64.encodeUrl(raw[0:]);
    u8 h[32];
    tokenHash(s.token[0:], h);
    s.expiresAt = pg.queryOne<String>(db,
        "INSERT INTO sessions (token_hash, user_id, expires_at) VALUES ($1, $2, now() + interval '30 days')"
        " RETURNING to_char(expires_at AT TIME ZONE 'UTC', 'YYYY-MM-DD\"T\"HH24:MI:SS\"Z\"')",
        h[0:32], found.data[0].id);
    w.json(201, reflect.of(s));
}

/* Is token a live session? Then the handlers get its user (userOf) */
bool checkToken(u8[] token, http.Request *r)
{
    pg.Pool *db = r.app;
    u8 h[32];
    tokenHash(token, h);
    vec.Vec<String> ids = pg.queryAll<String>(db,
        "SELECT user_id::text FROM sessions WHERE token_hash = $1 AND expires_at > now()", h[0:32]);
    defer pg.free(reflect.of(ids));
    if (!ids.len) { return false; }
    r.setText("userId", ids.data[0][0:]);         /* a copy, kept until the reply */
    return true;
}

/* The signed-in user's id (behind middleware.bearer) */
u8[] userOf(http.Request *r)
{
    return r.getText("userId");
}

void logout(http.Writer *w, http.Request *r)
{
    pg.Pool *db = r.app;
    u8 h[32];
    tokenHash(str.trim(r.header("Authorization")[7:]), h);
    db.exec("DELETE FROM sessions WHERE token_hash = $1", h[0:32]);
    w.status(204);
}

/* GET /collections/{id}: its owner, or anyone when it is public */
void showCollection(http.Writer *w, http.Request *r)
{
    pg.Pool *db = r.app;
    uuid.UUID id;
    if (!uuid.parse(r.param("id"), &id)) {
        w.error(404, "no such collection");
        return;
    }
    /* someone else's private collection is a 404, not a 403: it doesn't
     * say that it exists */
    vec.Vec<Collection> found = pg.queryAll<Collection>(db,
        "SELECT id, name, public FROM collections WHERE id = $1 AND (public OR user_id::text = $2)",
        r.param("id"), maybeUser(r));
    defer pg.free(reflect.of(found));
    if (!found.len) {
        w.error(404, "no such collection");
        return;
    }
    Collection *c = &found.data[0];
    c.bookmarks = pg.queryAll<Bookmark>(db,
        "SELECT " COLUMNS " FROM bookmarks WHERE id IN"
        " (SELECT bookmark_id FROM collection_items WHERE collection_id = $1) ORDER BY id DESC", r.param("id"));
    w.json(200, reflect.of(*c));
}

/* GET /bookmarks?tag=go&q=fast+servers&limit=20&after=<cursor>: newest first */
void list(http.Writer *w, http.Request *r)
{
    pg.Pool *db = r.app;
    i64 limit = 20;
    if (r.hasQuery("limit") && (!str.toInt(r.query("limit"), &limit) || limit < 1 || limit > 100)) {
        w.error(400, "limit is 1 to 100");
        return;
    }
    uuid.UUID after;
    u8[] cursor = r.query("after");
    if (cursor.len && !uuid.parse(cursor, &after)) {
        w.error(400, "after isn't a cursor from this API");
        return;
    }
    Page page;
    defer page.freeDecoded();
    /* one more than asked for: if it comes back, there is a next page */
    page.items = pg.queryAll<Bookmark>(db,
        "SELECT " COLUMNS " FROM bookmarks"
        " WHERE user_id = $1"
        "   AND ($2 = '' OR id IN (SELECT bookmark_id FROM tags WHERE tag = $2))"
        "   AND ($3 = '' OR search @@ websearch_to_tsquery('simple', $3))"
        "   AND ($4 = '' OR id < $4::uuid)"
        " ORDER BY id DESC LIMIT $5",
        userOf(r), r.query("tag"), r.query("q"), cursor, limit + 1);
    if (page.items.len > limit) {
        Bookmark extra = page.items.removeAt(limit);
        pg.free(reflect.of(extra));
        page.next = str.from(page.items.data[limit - 1].id[0:]);
    }
    w.json(200, reflect.of(page));
}

/* A tag is 1 to 30 lowercase letters, digits and dashes */
bool goodTag(u8[] t)
{
    if (t.len < 1 || t.len > 30) { return false; }
    for (auto c : t) {
        if (!((c >= 'a' && c <= 'z') || (c >= '0' && c <= '9') || c == '-')) { return false; }
    }
    return true;
}

/* The id in the path, or false (and a 404) when it isn't a UUID */
bool pathId(http.Writer *w, http.Request *r, uuid.UUID *id)
{
    if (uuid.parse(r.param("id"), id)) { return true; }
    w.error(404, "no such bookmark");
    return false;
}

void get(http.Writer *w, http.Request *r)
{
    pg.Pool *db = r.app;
    uuid.UUID id;
    if (!pathId(w, r, &id)) { return; }
    Bookmark b;
    try {
        b = pg.queryOne<Bookmark>(db, "SELECT " COLUMNS " FROM bookmarks WHERE id = $1 AND user_id = $2",
                                  r.param("id"), userOf(r));
    } catch (Error e) {
        if (e.code != pg.ERR_NO_ROWS) { throw(e); }
        w.error(404, "no such bookmark");
        return;
    }
    defer b.freeDecoded();
    w.json(200, reflect.of(b));
}

void create(http.Writer *w, http.Request *r)
{
    pg.Pool *db = r.app;
    NewBookmark in = http.decodeValid<NewBookmark>(r);
    if (in.tags.len > 10) {
        w.error(422, "at most 10 tags");
        return;
    }
    for (auto t : in.tags) {
        if (!goodTag(t[0:])) {
            w.error(422, "a tag is 1 to 30 lowercase letters, digits and dashes");
            return;
        }
    }
    String id = uuid.v7().string();
    defer id.free();
    /* the bookmark and its tags together, or neither */
    pg.Tx tx = db.begin();
    defer tx.rollback();
    try {
        tx.exec("INSERT INTO bookmarks (id, user_id, url, title, notes) VALUES ($1, $2, $3, $4, $5)",
                id, userOf(r), in.url, in.title.len ? in.title : in.url, in.notes);
    } catch (Error e) {
        if (e.code != pg.UNIQUE_VIOLATION) { throw(e); }
        w.error(409, "that link is already on the shelf");
        return;
    }
    tx.exec("INSERT INTO tags (bookmark_id, tag) SELECT $1, unnest($2::text[]) ON CONFLICT DO NOTHING", id, in.tags);
    Bookmark b = pg.queryOne<Bookmark>(&tx, "SELECT " COLUMNS " FROM bookmarks WHERE id = $1", id);
    defer b.freeDecoded();
    tx.commit();
    w.json(201, reflect.of(b));
}

void remove(http.Writer *w, http.Request *r)
{
    pg.Pool *db = r.app;
    uuid.UUID id;
    if (!pathId(w, r, &id)) { return; }
    if (db.exec("DELETE FROM bookmarks WHERE id = $1 AND user_id = $2", r.param("id"), userOf(r)) == 0) {
        w.error(404, "no such bookmark");
        return;
    }
    w.status(204);
}

/* shelf migrate up | down | status */
i32 runMigrate(migrate.Migrations *ms, migrate.Driver *d, u8[] what)
{
    if (str.equals(what, "up")) {
        printf("%d migrations ran\n", ms.up(d));
    } else if (str.equals(what, "down")) {
        printf("%d migration reverted\n", ms.down(d, 1));
    } else if (str.equals(what, "status")) {
        vec.Vec<migrate.Status> all = ms.status(d);
        for (auto s : all) {
            printf("%4d  %-22s %s\n", s.version, s.name, s.applied ? "applied" : "pending");
        }
        all.free();
    } else {
        log.error("shelf migrate: up, down or status, not `%s`", what);
        return 2;
    }
    return 0;
}

struct Settings {
    i64 port;
    bool accessLog;
    @required u8[] databaseUrl;
};

Settings gSettings;

i32 main(i32 argc, u8 **argv)
{
    gSettings.port = 8080;
    config.loadDotenv(".env");
    try {
        config.fromEnv(reflect.of(gSettings), "SHELF_");
    } catch (Error e) {
        log.error("%s", e.message);     /* every bad or missing variable, by name */
        os.exit(2);
    }

    flag.Set *flags = flag.new("shelf", "a bookmarks API");
    defer flags.free();
    i64 *port = flags.integer("port", gSettings.port, "the port to listen on", 'p');
    bool *accessLog = flags.boolean("access-log", gSettings.accessLog, "log every request");
    flag.Set *migrateCmd = flags.command("migrate", "up, down or status, then exit");
    flags.parseOrExit(argc, argv);

    /* the database: a pool of connections shared by every thread */
    pg.Pool *db = pg.open(gSettings.databaseUrl);
    defer db.close();
    migrate.Migrations ms = migrate.load("migrations");
    defer ms.free();
    migrate.Driver d = postgres.driverPool(db);
    if (flags.chosen() == migrateCmd) {
        return runMigrate(&ms, &d, migrateCmd.arg(0));
    }
    i64 ran = ms.up(&d);
    log.info("database at version %d (%d migrations ran)", migrate.version(&d), ran);

    String noise = uuid.v4().string();
    gNoUser = crypto.hashPassword(noise[0:]);
    noise.free();

    http.Mux mux;
    mux.handle("POST /users", &signup);
    mux.handle("POST /sessions", &login);
    http.Mux *session = mux.group("/sessions");
    middleware.bearer(session, &checkToken);
    session.handle("DELETE", &logout);

    /* every bookmark route needs a signed-in user */
    http.Mux *bookmarks = mux.group("/bookmarks");
    middleware.bearer(bookmarks, &checkToken);
    bookmarks.handle("GET", &list);
    bookmarks.handle("POST", &create);
    bookmarks.handle("GET /{id}", &get);
    bookmarks.handle("DELETE /{id}", &remove);

    http.Mux *collections = mux.group("/collections");
    middleware.bearer(collections, &checkToken);
    collections.handle("POST", &newCollection);
    collections.handle("PUT /{id}/bookmarks/{bookmark}", &addToCollection);
    mux.handle("GET /collections/{id}", &showCollection);      /* outside the group: anyone may ask */

    http.Options o;
    o.accessLog = *accessLog;
    log.info("Shelf listening on http://localhost:%d", *port);
    return http.listen((i32)*port, &mux, db, &o);
}
