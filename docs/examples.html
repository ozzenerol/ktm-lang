    Example projects

    Three complete programs in docs/examples/, each with the exact output it should produce. They double as tests: docs/examples/run.sh builds every project, runs it both compiled and in the JIT, and compares the output with its expected.txt.

    docs/examples/run.sh          # all projects
docs/examples/run.sh life     # just one

    Word frequency#

    Reads a text file and prints its most common words. It shows reading a file, scanning characters, counting with a StrMap, ranking with PtrVecSort, and reading command-line arguments.

    cd docs/examples/wordfreq
ktm main.ktm -o wordfreq && ./wordfreq sample.txt
    // Word frequency: reads a text file and prints its most common words.
//
//   ktm main.ktm -o wordfreq && ./wordfreq sample.txt
//   ktm -jit main.ktm sample.txt
//
// Shows: reading files, scanning strings, StrMap, PtrVec sorting,
// classes and command-line arguments.

#define TOP_N    8
#define MAX_WORD 64

class WordCount
{
  U8 *word;
  I64 count;
};

// Sort predicate: TRUE when a ranks above b. Most frequent first, then
// alphabetical so ties have a stable order.
I64 RanksHigher(U0 *_a, U0 *_b)
{
  WordCount *a = _a;
  WordCount *b = _b;
  if (a->count != b->count) {
    return a->count > b->count;
  }
  return StrCmp(a->word, b->word) < 0;
}

Bool IsLetter(U8 c)
{
  return ('a' <= c <= 'z') || ('A' <= c <= 'Z');
}

U0 CountWord(StrMap *counts, U8 *word)
{
  WordCount *wc = StrMapGet(counts, word);
  if (wc == NULL) {
    wc = MAlloc(sizeof(WordCount));
    wc->word = StrNew(word);       // the map keeps this pointer as its key
    wc->count = 0;
    StrMapSet(counts, wc->word, wc);
  }
  wc->count++;
}

I32 Main(I32 argc, U8 **argv)
{
  U8 *path = "sample.txt";
  if (argc > 1) {
    path = argv[1];
  }

  I64 size;
  U8 *text = FileRead(path, &size);
  if (text == NULL) {
    "cannot read %s\n", path;
    return 1;
  }

  StrMap *counts = StrMapNew();
  U8 word[MAX_WORD];
  I64 len = 0, total = 0;

  // One pass over the text; i == size flushes the final word.
  for (I64 i = 0; i <= size; i++) {
    U8 c = 0;
    if (i < size) c = text[i];
    if (IsLetter(c)) {
      if (len < MAX_WORD - 1) {
        word[len++] = ToLower(c);
      }
    } else if (len > 0) {
      word[len] = 0;
      CountWord(counts, word);
      total++;
      len = 0;
    }
  }

  PtrVec *ranked = PtrVecNew();
  I64 idx = 0;
  WordCount *wc;
  while (StrMapValueIter(counts, &idx, &wc)) {
    PtrVecPush(ranked, wc);
  }
  PtrVecSort(ranked, &RanksHigher);

  "%d words, %d distinct\n", total, ranked->size;
  for (I64 i = 0; i < TOP_N && i < ranked->size; i++) {
    WordCount *top = ranked->entries[i];
    "%2d. %-10s %d\n", i + 1, top->word, top->count;
  }

  Free(text);
  return 0;
}

    Bank#

    Accounts with monthly interest and a transaction log. A savings account inherits from Account and swaps in its own interest function through a function pointer. Errors are exceptions, and a failed transfer rolls itself back. Money is kept in integer cents, so no rounding errors creep in.

    cd docs/examples/bank
ktm main.ktm -o bank && ./bank
    // Bank: accounts, monthly interest and a transaction log.
//
//   ktm main.ktm -o bank && ./bank
//   ktm -jit main.ktm
//
// Shows: classes and inheritance, function pointers as "methods",
// exceptions for errors, linked lists, and integer money (cents).

#define ERR_FUNDS  'Funds'
#define ERR_FROZEN 'Frozen'

class Account
{
  U8 *owner;
  I64 balance;          // in cents
  Bool frozen;
  // Per-kind behaviour: interest earned in a month.
  I64 (*interest)(Account *a);
};

class Savings : Account
{
  I64 rate_bp;          // annual rate in basis points (1/100 of a percent)
};

class Txn
{
  U8 *owner;
  U8 *what;
  I64 amount;
};

List *ledger;

I64 NoInterest(Account *a)
{
  return 0;
}

I64 SavingsInterest(Account *a)
{
  Savings *s = a;
  return s->balance * s->rate_bp / 10000 / 12;
}

U0 InitAccount(Account *a, U8 *owner, I64 cents)
{
  a->owner = owner;
  a->balance = cents;
  a->frozen = FALSE;
  a->interest = &NoInterest;
}

U0 Record(Account *a, U8 *what, I64 amount)
{
  Txn *t = MAlloc(sizeof(Txn));
  t->owner = a->owner;
  t->what = what;
  t->amount = amount;
  ListAppend(ledger, t);
}

U0 Withdraw(Account *a, I64 cents)
{
  if (a->frozen) throw(ERR_FROZEN);
  if (cents > a->balance) throw(ERR_FUNDS);
  a->balance -= cents;
  Record(a, "withdraw", -cents);
}

U0 Deposit(Account *a, I64 cents)
{
  if (a->frozen) throw(ERR_FROZEN);
  a->balance += cents;
  Record(a, "deposit", cents);
}

// Moves money between accounts, or neither changes.
Bool Transfer(Account *from, Account *to, I64 cents)
{
  Bool withdrawn = FALSE;
  try {
    Withdraw(from, cents);
    withdrawn = TRUE;
    Deposit(to, cents);
  } catch {
    if (withdrawn) {                 // the deposit failed: put it back
      from->balance += cents;
      Record(from, "refund", cents);
    }
    U8 *why = "account frozen";
    if (Fs->except_ch == ERR_FUNDS) why = "insufficient funds";
    "  transfer %s -> %s refused: %s\n", from->owner, to->owner, why;
    return FALSE;
  }
  return TRUE;
}

U0 PrintMoney(I64 cents)
{
  if (cents < 0) {
    "-";
    cents = -cents;
  }
  "%d.%02d", cents / 100, cents % 100;
}

U0 Statement(Account *a)
{
  "  %-6s ", a->owner;
  PrintMoney(a->balance);
  if (a->frozen) " (frozen)";
  "\n";
}

U0 Main()
{
  ledger = ListNew();

  Account alice;
  InitAccount(&alice, "alice", 120000);

  Savings bob;
  InitAccount(&bob, "bob", 50000);
  bob.rate_bp = 450;                 // 4.5% a year
  bob.interest = &SavingsInterest;

  Account carol;
  InitAccount(&carol, "carol", 2500);

  Account *all[3];
  all[0] = &alice;
  all[1] = &bob;
  all[2] = &carol;

  "Opening balances\n";
  for (I64 i = 0; i < 3; i++) Statement(all[i]);

  "Transfers\n";
  Transfer(&alice, &bob, 30000);
  Transfer(&carol, &alice, 9999);    // carol only has 25.00
  carol.frozen = TRUE;
  Transfer(&bob, &carol, 100);

  "Month end\n";
  for (I64 i = 0; i < 3; i++) {
    Account *a = all[i];
    I64 earned = a->interest(a);
    if (earned > 0) {
      a->balance += earned;
      Record(a, "interest", earned);
    }
    Statement(a);
  }

  "Ledger (%d entries)\n", ListCount(ledger);
  for (List *it = ledger->next; it != ledger; it = it->next) {
    Txn *t = it->value;
    "  %-6s %-9s ", t->owner, t->what;
    PrintMoney(t->amount);
    "\n";
  }
}

    Game of Life#

    Conway's Game of Life on a small board that wraps at the edges, with a glider and a blinker. It shows 2D arrays, nested loops, % for wrap-around and MemCpy on whole arrays.

    cd docs/examples/life
ktm main.ktm -o life && ./life
    // Conway's Game of Life on a small wrapping board.
//
//   ktm main.ktm -o life && ./life
//   ktm -jit main.ktm
//
// Shows: 2D arrays, nested loops, the % operator for wrap-around,
// #define constants and building output a line at a time.

#define W     12
#define H     8
#define STEPS 4

U8 board[H][W];
U8 next[H][W];

U0 Place(U8 *pattern, I64 top, I64 left)
{
  // `pattern` rows are separated by '/'; '#' is a live cell.
  I64 r = top, c = left;
  for (U8 *p = pattern; *p; p++) {
    if (*p == '/') {
      r++;
      c = left;
    } else {
      board[r][c] = *p == '#';
      c++;
    }
  }
}

I64 Neighbours(I64 r, I64 c)
{
  I64 n = 0;
  for (I64 dr = -1; dr <= 1; dr++) {
    for (I64 dc = -1; dc <= 1; dc++) {
      if (dr == 0 && dc == 0) continue;
      n += board[(r + dr + H) % H][(c + dc + W) % W];
    }
  }
  return n;
}

U0 Step()
{
  for (I64 r = 0; r < H; r++) {
    for (I64 c = 0; c < W; c++) {
      I64 n = Neighbours(r, c);
      // Born with 3 neighbours, survives with 2 or 3.
      next[r][c] = n == 3 || (board[r][c] && n == 2);
    }
  }
  MemCpy(board, next, sizeof(board));
}

I64 Population()
{
  I64 alive = 0;
  for (I64 r = 0; r < H; r++) {
    for (I64 c = 0; c < W; c++) {
      alive += board[r][c];
    }
  }
  return alive;
}

U0 Show(I64 gen)
{
  "generation %d, %d alive\n", gen, Population;
  U8 line[W + 1];
  line[W] = 0;
  for (I64 r = 0; r < H; r++) {
    for (I64 c = 0; c < W; c++) {
      line[c] = '.';
      if (board[r][c]) line[c] = '#';
    }
    "%s\n", line;
  }
}

U0 Main()
{
  Place(".#./..#/###", 0, 0);   // a glider, top left
  Place("###", 5, 7);           // a blinker, bottom right
  for (I64 gen = 0; gen <= STEPS; gen++) {
    Show(gen);
    Step;
  }
}

    More examples in the repository#

    The compiler's own test suite, src/tests/, has more than 300 small programs, one per feature or fixed bug: threads, sockets, JSON, SQLite, fuzzy matching and more. The standard library in src/std/ is all ktm too, and worth reading.
