    Example projects

    Three complete programs in docs/examples/, each with the exact output it should produce. They double as tests: docs/examples/run.sh builds every project, runs it both compiled and in the JIT, and compares the output with its expected.txt.

    docs/examples/run.sh          # all projects
docs/examples/run.sh life     # just one

    Word frequency#

    Reads a text file and prints its most common words. It shows reading a file, scanning characters, counting with a map.Map, ranking with a vec.Vec's sort, and reading command-line arguments.

    cd docs/examples/wordfreq
ktm main.ktm -o wordfreq && ./wordfreq sample.txt
    // Word frequency: reads a text file and prints its most common words.
//
//   ktm main.ktm -o wordfreq && ./wordfreq sample.txt
//   ktm -jit main.ktm sample.txt
//
// Shows: reading a file (fs), scanning strings, counting in a map.Map,
// sorting a vec.Vec, structs and command-line arguments.

import cstr, fs, map, str, vec;

const i64 TOP_N = 8;
const i64 MAX_WORD = 64;

struct WordCount {
    u8[] word;
    i64 count;
};

// Most frequent first, then alphabetical so ties have a stable order.
i64 byRank(WordCount *a, WordCount *b) {
    if (a.count != b.count) {
        return b.count - a.count;
    }
    return str.compare(a.word, b.word);
}

bool isLetter(u8 c) {
    return ('a' <= c <= 'z') || ('A' <= c <= 'Z');
}

i32 main(i32 argc, u8 **argv) {
    u8[] path = "sample.txt";
    if (argc > 1) {
        path = cstr.view(argv[1]);
    }

    ad String text = fs.readAll(path);
    if (!text.ptr) {
        printf("cannot read %.*s\n", (i32)path.len, path.ptr);
        return 1;
    }
    i64 size = text.len;

    ad map.Map<u8[], i64> counts = map.new<u8[], i64>();
    u8 word[MAX_WORD];
    i64 len = 0, total = 0;

    // One pass over the text; i == size flushes the final word.
    for (i64 i = 0; i <= size; i++) {
        u8 c = 0;
        if (i < size) { c = text[i]; }
        if (isLetter(c)) {
            if (len < MAX_WORD - 1) {
                word[len++] = str.toLower(c);
            }
        } else if (len > 0) {
            counts.set(word[0:len], counts.getOr(word[0:len], 0) + 1);   // the map copies the key
            total++;
            len = 0;
        }
    }

    ad vec.Vec<WordCount> ranked = vec.new<WordCount>(counts.len + 1);
    i64 it = 0;
    WordCount wc;
    while (counts.next(&it, &wc.word, &wc.count)) {
        ranked.push(wc);
    }
    ranked.sort(&byRank);

    printf("%d words, %d distinct\n", total, ranked.len);
    for (i64 i = 0; i < TOP_N && i < ranked.len; i++) {
        WordCount top = ranked.get(i);
        printf("%2d. %-10.*s %d\n", i + 1, (i32)top.word.len, top.word.ptr, top.count);
    }

    return 0;
}

    Bank#

    Accounts with monthly interest and a transaction log. A savings account inherits from Account and swaps in its own interest function through a function pointer. Errors are exceptions, and a failed transfer rolls itself back. Money is kept in integer cents, so no rounding errors creep in.

    cd docs/examples/bank
ktm main.ktm -o bank && ./bank
    // Bank: accounts, monthly interest and a transaction log.
//
//   ktm main.ktm -o bank && ./bank
//   ktm -jit main.ktm
//
// Shows: structs and inheritance, function pointers in a struct,
// exceptions with an error enum, the list module, and integer money (cents).

import list, mem;

enum BankError { funds = 1, frozen };

struct Account {
    u8 *owner;
    i64 balance;          // in cents
    bool frozen;
    // Per-kind behaviour: interest earned in a month.
    fn(Account *a) i64 interest;
};

struct Savings : Account {
    i64 rateBp;          // annual rate in basis points (1/100 of a percent)
};

struct Txn {
    u8 *owner;
    u8 *what;
    i64 amount;
};

list.List<Txn *> ledger;

i64 noInterest(Account *a) {
    return 0;
}

i64 savingsInterest(Account *a) {
    Savings *s = a;
    return s.balance * s.rateBp / 10000 / 12;
}

void initAccount(Account *a, u8 *owner, i64 cents) {
    a.owner = owner;
    a.balance = cents;
    a.frozen = false;
    a.interest = &noInterest;
}

void record(Account *a, u8 *what, i64 amount) {
    Txn *t = mem.alloc(sizeof(Txn));
    t.owner = a.owner;
    t.what = what;
    t.amount = amount;
    ledger.append(t);
}

void withdraw(Account *a, i64 cents) {
    if (a.frozen) { throw(BankError.frozen); }
    if (cents > a.balance) { throw(BankError.funds); }
    a.balance -= cents;
    record(a, "withdraw", -cents);
}

void deposit(Account *a, i64 cents) {
    if (a.frozen) { throw(BankError.frozen); }
    a.balance += cents;
    record(a, "deposit", cents);
}

// Moves money between accounts, or neither changes.
bool transfer(Account *from, Account *to, i64 cents) {
    bool withdrawn = false;
    try {
        withdraw(from, cents);
        withdrawn = true;
        deposit(to, cents);
    } catch (Error e) {
        if (withdrawn) {                 // the deposit failed: put it back
            from.balance += cents;
            record(from, "refund", cents);
        }
        u8 *why = "account frozen";
        if (e.code == BankError.funds) { why = "insufficient funds"; }
        printf("  transfer %s -> %s refused: %s\n", from.owner, to.owner, why);
        return false;
    }
    return true;
}

void printMoney(i64 cents) {
    if (cents < 0) {
        printf("-");
        cents = -cents;
    }
    printf("%d.%02d", cents / 100, cents % 100);
}

void statement(Account *a) {
    printf("  %-6s ", a.owner);
    printMoney(a.balance);
    if (a.frozen) { printf(" (frozen)"); }
    printf("\n");
}

void main() {
    ledger = list.new<Txn *>();

    Account alice;
    initAccount(&alice, "alice", 120000);

    Savings bob;
    initAccount(&bob, "bob", 50000);
    bob.rateBp = 450;                 // 4.5% a year
    bob.interest = &savingsInterest;

    Account carol;
    initAccount(&carol, "carol", 2500);

    Account *all[3];
    all[0] = &alice;
    all[1] = &bob;
    all[2] = &carol;

    printf("Opening balances\n");
    for (i64 i = 0; i < 3; i++) { statement(all[i]); }

    printf("Transfers\n");
    transfer(&alice, &bob, 30000);
    transfer(&carol, &alice, 9999);    // carol only has 25.00
    carol.frozen = true;
    transfer(&bob, &carol, 100);

    printf("Month end\n");
    for (i64 i = 0; i < 3; i++) {
        Account *a = all[i];
        i64 earned = a.interest(a);
        if (earned > 0) {
            a.balance += earned;
            record(a, "interest", earned);
        }
        statement(a);
    }

    printf("Ledger (%d entries)\n", ledger.len);
    for (auto n = ledger.head; n; n = n.next) {
        Txn *t = n.value;
        printf("  %-6s %-9s ", t.owner, t.what);
        printMoney(t.amount);
        printf("\n");
    }
    for (auto n = ledger.head; n; n = n.next) { mem.free(n.value); }
    ledger.free();
}

    Game of Life#

    Conway's Game of Life on a small board that wraps at the edges, with a glider and a blinker. It shows 2D arrays, nested loops, % for wrap-around and mem.copy on whole arrays.

    cd docs/examples/life
ktm main.ktm -o life && ./life
    // Conway's Game of Life on a small wrapping board.
//
//   ktm main.ktm -o life && ./life
//   ktm -jit main.ktm
//
// Shows: 2D arrays, nested loops, the % operator for wrap-around,
// constants and building output a line at a time.

import mem;

const i64 W = 12;
const i64 H = 8;
const i64 STEPS = 4;

u8 board[H][W];
u8 next[H][W];

void place(u8 *pattern, i64 top, i64 left) {
    // `pattern` rows are separated by '/'; '#' is a live cell.
    i64 r = top, c = left;
    for (u8 *p = pattern; *p; p++) {
        if (*p == '/') {
            r++;
            c = left;
        } else {
            board[r][c] = *p == '#';
            c++;
        }
    }
}

i64 neighbours(i64 r, i64 c) {
    i64 n = 0;
    for (i64 dr = -1; dr <= 1; dr++) {
        for (i64 dc = -1; dc <= 1; dc++) {
            if (dr == 0 && dc == 0) { continue; }
            n += board[(r + dr + H) % H][(c + dc + W) % W];
        }
    }
    return n;
}

void step() {
    for (i64 r = 0; r < H; r++) {
        for (i64 c = 0; c < W; c++) {
            i64 n = neighbours(r, c);
            // Born with 3 neighbours, survives with 2 or 3.
            next[r][c] = n == 3 || (board[r][c] && n == 2);
        }
    }
    mem.copy(board, next, sizeof(board));
}

i64 population() {
    i64 alive = 0;
    for (i64 r = 0; r < H; r++) {
        for (i64 c = 0; c < W; c++) {
            alive += board[r][c];
        }
    }
    return alive;
}

void show(i64 gen) {
    printf("generation %d, %d alive\n", gen, population());
    u8 line[W + 1];
    line[W] = 0;
    for (i64 r = 0; r < H; r++) {
        for (i64 c = 0; c < W; c++) {
            line[c] = '.';
            if (board[r][c]) { line[c] = '#'; }
        }
        printf("%s\n", line);
    }
}

void main() {
    place(".#./..#/###", 0, 0);   // a glider, top left
    place("###", 5, 7);           // a blinker, bottom right
    for (i64 gen = 0; gen <= STEPS; gen++) {
        show(gen);
        step();
    }
}
