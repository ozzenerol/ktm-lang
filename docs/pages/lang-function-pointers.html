Function pointers

A function pointer type is written fn(params) Ret. No C declarator puzzles.

At a glance

fn(i64) i64 f = &twice;          // a variable
f(5);                            // call it
fn(i64) i64 ops[] = {&a, &b};    // an array of them
fn(i64) i64 pick(bool b);        // a function returning one
(fn(i64) i64) *pf = &f;          // a pointer to one: parentheses
type Handler = fn(u8[]) bool;    // give it a name

C

int (*op)(int, int) = &add;
int (*ops[2])(int) = {...};
int (*pick(int))(int);

ktm

fn(i32, i32) i32 op = &add;
fn(i32) i32 ops[2] = {...};
fn(i32) i32 pick(i32);

Passing a function


i64 add(i64 a, i64 b) { return a + b; }
i64 mul(i64 a, i64 b) { return a * b; }

i64 fold(fn(i64, i64) i64 op, i64[] nums)
{
  i64 acc = nums[0];
  for (i64 i = 1; i < nums.len; i++) { acc = op(acc, nums[i]); }
  return acc;
}

void main()
{
  i64 nums[] = {1, 2, 3, 4};
  printf("%d %d\n", fold(&add, nums), fold(&mul, nums));
}

  • Parameter names are optional and ignored: fn(i64 a, i64 b) i64 is fn(i64, i64) i64.
  • (*f)(x) calls it too, as in C.
  • The arguments are checked against the pointer's parameter types, like a direct call.

Tables of functions

The array size goes after the name, as in C. Locals, globals, fields and parameters all work.


i64 dbl(i64 x) { return x * 2; }
i64 sq(i64 x)  { return x * x; }
i64 neg(i64 x) { return -x; }

fn(i64) i64 ops[] = {&dbl, &sq, &neg};
u8 *names[] = {"double", "square", "negate"};

void main()
{
  for (i64 i = 0; i < 3; i++) {
    printf("%s(5) = %d\n", names[i], ops[i](5));
  }
}

double(5) = 10
square(5) = 25
negate(5) = -5

In a struct

A field can be a function pointer. A default in its parameter list fills a left-out argument.


i64 dbl(i64 x) { return x * 2; }

struct Calc { u8 *name; fn(i64 x = 21) i64 op; };

void main()
{
  Calc c = {"doubler", &dbl};
  printf("%s: %d %d\n", c.name, c.op(5), c.op());
}

doubler: 10 42

Returning a function


i64 dbl(i64 x) { return x * 2; }
i64 sq(i64 x)  { return x * x; }

fn(i64) i64 choose(bool square)
{
  if (square) { return &sq; }
  return &dbl;
}

void main()
{
  fn(i64) i64 f = choose(true);
  printf("%d %d\n", f(5), choose(false)(5));
}

A pointer to a function pointer

Put the fn type in parentheses: (fn(i64) i64) *.


i64 dbl(i64 x) { return x * 2; }
i64 sq(i64 x)  { return x * x; }

void swapIn((fn(i64) i64) *slot, fn(i64) i64 g) { *slot = g; }

void main()
{
  fn(i64) i64 f = &sq;
  printf("%d ", f(5));
  swapIn(&f, &dbl);            // changes f through the pointer
  printf("%d %s\n", f(5), typeof(&f));
}

Calling any expression

Anything whose type is a function pointer can be called: a cast, a call's result, an assignment.


import cstr;

i64 add(i64 a, i64 b) { return a + b; }
i64 mul(i64 a, i64 b) { return a * b; }

void *lookup(u8 *name)
{
  if (!cstr.cmp(name, "add")) { return &add; }
  return &mul;
}

void main()
{
  fn(i64, i64) i64 f;
  printf("%d\n", ((fn(i64, i64) i64)lookup("add"))(2, 3));
  printf("%d\n", (f = &mul)(4, 5));
}

Defaults and varargs

A function pointer's parameters can have defaults and take extras in every form a function does.


i64 add(i64 a, i64 b) { return a + b; }
i64 count(...args) { return args.len; }

void main()
{
  fn(i64 a, i64 b = 10) i64 k = &add;
  fn(...args) i64 c = &count;
  printf("%d %d %d\n", k(1), k(1, 2), c(4, 5, 6));
}

Callbacks with real types

A void * parameter on one side matches any pointer parameter on the other. So a comparator takes the types it really gets, and is passed where fn(void *, void *) i64 is expected, without a cast.


struct Item { i64 key; };

i64 itemLess(Item *a, Item *b) { return a.key < b.key; }

void main()
{
  Item a = {1}, b = {2};
  fn(void *, void *) i64 cmp = &itemLess;
  printf("%d %d\n", cmp(&a, &b), cmp(&b, &a));
}

A void * and null convert to and from a function pointer without a cast:


i64 dbl(i64 x) { return x * 2; }

void main()
{
  void *raw = &dbl;
  fn(i64) i64 back = raw;
  fn(i64) i64 none = null;
  printf("%d %d %d\n", back(4), raw == &dbl, none == null);
}

What doesn't convert

Anything else is an error that names both types:

i64 t(i64 x) { return x; }
fn(u8 *) i64 op = &t;    // error: cannot convert fn(i64) i64 to fn(u8*) i64

These need a cast:

  • a different return type (returning Item * is not fn(void *) void *);
  • different integer, struct, slice or function pointer parameters;
  • a pointer to a function pointer: (fn(Item *) i64) * is not (fn(void *) i64) *.

C's form, i64 (*f)(i64), is an error that shows the fn(...) spelling to use. fn is a keyword.
