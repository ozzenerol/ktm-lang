6. Migrations

schema.sql can create a table but not change one that exists. Real schemas change with the code, so they live in numbered migration files, and the database remembers which ones it has run.

The package

$ ktm get github.com/ozzenerol/ktm-migrate

ktm-migrate works with any SQL engine through a small driver; ktm-pg's comes with it as "github.com/ozzenerol/ktm-migrate/postgres".

The files

schema.sql becomes the first migration, and a second one adds a field for notes:

migrations/
  001_create_bookmarks.up.sql      the CREATE TABLE of chapter 5
  001_create_bookmarks.down.sql    DROP TABLE bookmarks;
  002_add_notes.up.sql             ALTER TABLE bookmarks ADD notes text NOT NULL DEFAULT '';
  002_add_notes.down.sql           ALTER TABLE bookmarks DROP notes;
  • The number is the version; files run in order, each once. .down.sql undoes its .up.sql.
  • Each migration runs in a transaction with the row that records it, so it happens whole or not at all.
  • Never edit a migration that has run somewhere: add a new one.

main

i32 main(i32 argc, u8 **argv)
{
    gSettings.port = 8080;
    config.loadDotenv(".env");
    config.fromEnv(reflect.of(gSettings), "SHELF_");
    validate.Errors errs;
    if (!validate.check(reflect.of(gSettings), &errs)) {
        String msg = errs.message();
        log.error("bad settings: %s", msg);
        os.exit(2);
    }
    errs.free();

    flag.Set *flags = flag.new("shelf", "a bookmarks API");
    defer flags.free();
    i64 *port = flags.integer("port", gSettings.port, "the port to listen on", 'p');
    bool *accessLog = flags.boolean("access-log", gSettings.accessLog, "log every request");
    flag.Set *migrateCmd = flags.command("migrate", "up, down or status, then exit");
    flags.parseOrExit(argc, argv);

    /* the database: a pool of connections shared by every thread */
    pg.Pool *db = pg.open(gSettings.databaseUrl);
    defer db.close();
    migrate.Migrations ms = migrate.load("migrations");
    defer ms.free();
    migrate.Driver d = postgres.driverPool(db);
    if (flags.chosen() == migrateCmd) {
        return runMigrate(&ms, &d, migrateCmd.arg(0));
    }
    i64 ran = ms.up(&d);
    log.info("database at version %d (%d migrations ran)", migrate.version(&d), ran);

    http.Mux mux;
    mux.handle("GET /bookmarks", &list);
    mux.handle("POST /bookmarks", &create);
    mux.handle("GET /bookmarks/{id}", &get);
    mux.handle("DELETE /bookmarks/{id}", &remove);

    http.Options o;
    o.accessLog = *accessLog;
    log.info("Shelf listening on http://localhost:%d", *port);
    return http.listen((i32)*port, &mux, db, &o);
}

At every start the server brings the database up to date before it listens, so a deploy is just "start the new binary". A lock in the database keeps two copies starting at once from both migrating. migrate is a subcommand (flag) for doing it by hand:

/* shelf migrate up | down | status */
i32 runMigrate(migrate.Migrations *ms, migrate.Driver *d, u8[] what)
{
    if (str.equals(what, "up")) {
        printf("%d migrations ran\n", ms.up(d));
    } else if (str.equals(what, "down")) {
        printf("%d migration reverted\n", ms.down(d, 1));
    } else if (str.equals(what, "status")) {
        vec.Vec<migrate.Status> all = ms.status(d);
        for (auto s : all) {
            printf("%4d  %-22s %s\n", s.version, s.name, s.applied ? "applied" : "pending");
        }
        all.free();
    } else {
        log.error("shelf migrate: up, down or status, not `%s`", what);
        return 2;
    }
    return 0;
}

$ ./shelf migrate status
   1  create_bookmarks       pending
   2  add_notes              pending

2 migrations ran
$ ./shelf migrate down
1 migration reverted

   1  create_bookmarks       applied
   2  add_notes              pending
