Structs and unions

A struct is a C struct whose name is a type straight away: no typedef. Methods and inheritance have their own page.

At a glance

struct Point { i64 x, y; };     // a type named Point
Point a = {1, 2};               // fields in order
Point *p = &a;  p.x = 5;        // . also works through a pointer
struct Node;                    // declared ahead, defined later
union Bits { i64 i; f64 f; };   // fields share memory

Defining and using


struct Point
{
  i64 x, y;
};

struct Player
{
  u8 *name;
  Point pos;
  i64 score;
};

void main()
{
  Point origin = {0, 0};
  Player p = {"ada", {3, 4}, 0};
  p.score += 10;
  p.pos.x = origin.x;
  printf("%s at (%d, %d) with %d\n", p.name, p.pos.x, p.pos.y, p.score);
}

ada at (0, 4) with 10

Values are copied

Structs are passed, returned and assigned by value.


struct Point { i64 x, y; };

Point make(i64 x, i64 y)
{
  Point r = {x, y};
  return r;
}

i64 sum(Point p)
{
  p.x = 1000;          // changes the copy only
  return p.x + p.y;
}

void main()
{
  Point a = make(1, 2);
  Point b = a;         // a copy
  b.x = 9;
  printf("%d %d %d\n", a.x, b.x, sum(a));
  printf("%d\n", a.x);
}

To change the caller's struct, pass a pointer:


struct Counter { i64 n; };

void bump(Counter *c) { c.n++; }   // c.n is c->n

void main()
{
  Counter c = {0};
  bump(&c);
  bump(&c);
  printf("%d\n", c.n);
}

Initialising

A list fills the fields in order; the rest are zero. Inner structs and arrays take nested lists.


struct Point { i64 x, y; };
struct Line  { Point from, to; i64 tags[2]; };

Line g = {{0, 0}, {3, 4}, {7, 8}};

void main()
{
  Line m = {{1, 2}, {5, 6}};       // tags are zero
  Point p = {7};                   // y is zero
  printf("%d %d %d\n", g.to.x, g.to.y, g.tags[1]);
  printf("%d %d %d | %d %d\n", m.from.y, m.to.x, m.tags[0], p.x, p.y);
}

3 4 8
2 5 0 | 7 0

A declared struct starts zeroed: after Point p; every field is 0, as with Point p = {0};.

Strings in fields

A string for a u8 array field is copied in and must fit with its NUL. A u8 * field points at the string instead.


struct Item { i64 id; u8 name[8]; u8 *note; };

u8 days[3][4] = {"Mon", "Tue", "Wed"};

void main()
{
  Item it = {1, "pen", "blue ink"};
  it.name[0] = 'P';             // our own copy: writable
  printf("%s %s %s %d\n", it.name, it.note, days[2], sizeof(Item));
}

Size and layout

The same layout as C: fields in order, each aligned to its size.


struct A { u8 tag; i64 value; };    // 7 bytes of padding after tag
struct B { i64 value; u8 tag; };    // padded at the end

void main()
{
  printf("%d %d %d\n", sizeof(A), sizeof(B), alignof(A));
}

16 16 8

Declaring ahead

struct Node; declares a struct defined later. It can be pointed to at once; it can be held by value only after its definition.


struct Node;

struct List { Node *head; i64 len; };

struct Node { i64 value; Node *next; };

void main()
{
  Node b = {2, null};
  Node a = {1, &b};
  List l = {&a, 2};
  for (Node *n = l.head; n; n = n.next) { printf("%d,", n.value); }
  printf(" len %d\n", l.len);
}

1,2, len 2

Holding it by value too early is the error field `b` of A holds a B, which is not complete here: use a pointer.

Unions

All fields of a union share the same memory.


union Bits
{
  i64 i;
  f64 f;
  u8 bytes[8];
};

void main()
{
  Bits b;
  b.i = 0x41;
  printf("%c %d\n", b.bytes[0], sizeof(Bits));
  b.f = 1.0;
  printf("%x\n", b.i);
}

3ff0000000000000

An initialiser list gives a union one item, for its first field.

Anonymous unions in a struct

Its fields are used as the struct's own. A tagged value is the classic use:


enum Kind { num, text };

struct Value
{
  Kind kind;
  union { i64 n; u8 *s; };
};

void show(Value v)
{
  switch (v.kind) {
    case Kind.num:  printf("number %d\n", v.n);
    case Kind.text: printf("text %s\n", v.s);
  }
}

void main()
{
  Value a = {Kind.num, 42};
  Value b;
  b.kind = Kind.text;
  b.s = "hi";
  show(a);
  show(b);
}

number 42
text hi

Named inner types

A struct or union defined inside a struct can have a tag and be used elsewhere:


struct Shape
{
  u8 kind;
  union Size { i32 side; f32 radius; };    // fields are Shape's own
  struct Pos { i32 x, y; } at;             // a field named at
};

void main()
{
  Shape s;
  s.side = 3;
  s.at.x = 10;
  Size z;
  z.radius = 1.5;
  printf("%d %d %d %d\n", s.side, s.at.x, sizeof(Size), sizeof(Shape));
}
